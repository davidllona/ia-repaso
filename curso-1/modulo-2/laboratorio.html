Antes de empezar: las piezas

Python es el lenguaje en el que se escribe todo. Sus órdenes se leen de arriba abajo, una por línea.

Python, solo, sabe hacer poco. Por eso se usan librerías: paquetes de código que otra gente ya escribió (para hacer cuentas, para tablas, para modelos). La orden import mete una librería en tu programa. Desde ese momento puedes usar sus herramientas.

Paso 1 · Calcular θ₀ y θ₁ con la fórmula de mínimos cuadrados

Para qué sirve Ver que la recta 115 + 34·x no es inventada: sale de una fórmula. Y calcular sus residuos y su MSE.

import numpy as np

x = np.array([1, 2, 3, 4, 5])
y = np.array([150, 190, 205, 250, 290])

theta1 = np.sum((x - x.mean()) * (y - y.mean())) / np.sum((x - x.mean()) ** 2)
theta0 = y.mean() - theta1 * x.mean()
print("theta0 =", theta0, "| theta1 =", theta1)

y_hat = theta0 + theta1 * x
residuos = y - y_hat
print("predicciones:", y_hat)
print("residuos:    ", residuos)
print("MSE =", np.mean(residuos ** 2))

Paso 2 · Inventar películas con presupuesto, estrellas y género

Para qué sirve Tener una tabla grande y realista para practicar sin internet. La receta escondida es: horas = 20 + 25 × presupuesto + 30 × estrellas + extra del género + azar.

import pandas as pd

rng = np.random.default_rng(42)
n = 400
presupuesto = np.round(rng.uniform(0.5, 6, n), 1)
estrellas = rng.integers(1, 6, n)
genero = rng.choice(["Comedia", "Accion", "Documental"], size=n)
extra = pd.Series(genero).map({"Comedia": 0, "Accion": 15, "Documental": -40})
horas = 20 + 25 * presupuesto + 30 * estrellas + extra + rng.normal(0, 12, n)

pelis = pd.DataFrame({"presupuesto": presupuesto, "estrellas": estrellas,
                      "genero": genero, "horas": horas.round(1)})
print(pelis.shape)
print(pelis.head())

def descenso(x, y, lr, pasos):
    cuota, precio = 0.0, 0.0
    for paso in range(1, pasos + 1):
        error = (cuota + precio * x) - y
        grad_cuota = 2 * np.mean(error)
        grad_precio = 2 * np.mean(error * x)
        cuota = cuota - lr * grad_cuota
        precio = precio - lr * grad_precio
        mse = np.mean(((cuota + precio * x) - y) ** 2)
        if paso <= 2 or paso == pasos:
            print(f"paso {paso:3}: cuota = {cuota:.3f} | precio = {precio:.3f} | MSE = {mse:.3f}")
    return cuota, precio

def descenso_lineal(X, y, lr, n_iter):
    """Gradient descent para y_hat = theta0 + theta1 * x, minimizando el MSE."""
    theta = np.zeros(2)                      # punto de partida: theta0 = 0, theta1 = 0
    n = len(y)
    thetas, perdidas = [], []
    for _ in range(n_iter):
        error = theta[0] + theta[1] * X - y  # y_hat - y de cada ejemplo
        thetas.append(theta.copy())
        perdidas.append(np.mean(error ** 2))
        grad0 = 2 / n * np.sum(error)        # derivada del MSE respecto a theta0
        grad1 = 2 / n * np.sum(error * X)    # derivada del MSE respecto a theta1
        theta = theta - lr * np.array([grad0, grad1])
    thetas.append(theta.copy())
    perdidas.append(np.mean((theta[0] + theta[1] * X - y) ** 2))
    return thetas, perdidas

def descenso_logistico(X, y, lr, n_iter):
    """Gradient descent para regresión logística. X ya lleva una columna de unos."""
    theta = np.zeros(X.shape[1])
    historial = []
    for _ in range(n_iter):
        p = 1 / (1 + np.exp(-X @ theta))                  # probabilidades con los pesos actuales
        historial.append(-np.mean(y * np.log(p) + (1 - y) * np.log(1 - p)))   # log-loss
        gradiente = X.T @ (p - y) / len(y)                # derivada del log-loss
        theta = theta - lr * gradiente
    return theta, historial

def minibatch(X, y, lr, epocas, batch, semilla=0):
    r = np.random.default_rng(semilla)
    theta = np.zeros(2)
    for _ in range(epocas):

# Entrena un modelo y devuelve cuántas iteraciones ha usado de verdad y el primer aviso (si lo hay)
def entrenar(modelo, X_tr, y_tr):
    with warnings.catch_warnings(record=True) as avisos:
        warnings.simplefilter("always")
        modelo.fit(X_tr, y_tr)
    return int(np.max(modelo.n_iter_)), (avisos[0] if avisos else None)

Para qué sirve Demostrar que añadir variables siempre sube el R² de entrenamiento, aunque sean basura, y que lo que de verdad importa es el R² en datos nuevos. Incluye el R² ajustado.
