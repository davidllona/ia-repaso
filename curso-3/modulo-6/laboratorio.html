Paso 2 · El retorno descontado, calculado de dos formas (fragmento 4)

Para qué sirve Calcular el retorno descontado de dos recomendadores (uno prudente y otro pesado) con tres valores de gamma, y comprobar que la cuenta «hacia atrás» da lo mismo que la fórmula directa.

import numpy as np

prudente  = [1, 1, 1, 1, 1, 10, 1, 1, 1000]
temerario = [1, 1, 1, 10, -100]

def retorno(recompensas, gamma):
    total = 0.0
    for t, r in enumerate(recompensas):
        total += gamma ** t * r
    return total

def retornos_hacia_atras(recompensas, gamma):
    G = 0.0
    salida = []
    for r in reversed(recompensas):
        G = r + gamma * G
        salida.append(G)
    return salida[::-1]

for gamma in [1.0, 0.9, 0.5]:
    print(f"gamma={gamma}: prudente={retorno(prudente, gamma):8.2f}  temerario={retorno(temerario, gamma):8.2f}")

print("G_t hacia atrás (prudente, gamma=0.9):", np.round(retornos_hacia_atras(prudente, 0.9), 2))
print("G_0 con la fórmula directa:            ", round(retorno(prudente, 0.9), 2))

def entrenar(algoritmo, episodios=500, semilla=0):

def crear_red():
    return keras.Sequential([keras.Input(shape=(4,)),
                             layers.Dense(64, activation="relu"),
                             layers.Dense(64, activation="relu"),
                             layers.Dense(2)])

def como_lote(lote):
    s, a, r, s2, f = zip(*lote)
    return (np.array(s, "float32"), np.array(a, "int32"), np.array(r, "float32"),
            np.array(s2, "float32"), np.array(f, "float32"))

@tf.function
def entrenar_lote(s, a, r, s2, f):
    y = r + gamma * tf.reduce_max(q_obj(s2), axis=1) * (1.0 - f)
    with tf.GradientTape() as cinta:
        q_sa = tf.reduce_sum(q(s) * tf.one_hot(a, 2), axis=1)
        perdida = huber(y, q_sa)
    grads = cinta.gradient(perdida, q.trainable_variables)
    optimizador.apply_gradients(zip(grads, q.trainable_variables))

@tf.function
def accion_greedy(estado):
    return tf.argmax(q(estado[None, :]), axis=1)[0]

def entrenar(esquema, episodios=5000, semilla=0, alpha=0.1, gamma=0.99):
    env = gym.make("FrozenLake-v1", is_slippery=True)
    rng = np.random.default_rng(semilla)

15 · Policy gradient mínimo: un paso de REINFORCE

Para qué sirve Ver con una sola transición qué hace el gradiente de política: si el retorno es positivo, sube la probabilidad de la acción hecha; si es negativo, la baja. Es la idea de REINFORCE, A2C y PPO reducida a un paso.

import tensorflow as tf
import keras
from keras import layers

                             layers.Dense(2, activation="softmax")])

accion = 1

def paso_reinforce(retorno):
    with tf.GradientTape() as cinta:
        probs = politica(estado)
        log_prob = tf.math.log(probs[0, accion])
        perdida = -log_prob * retorno
    grads = cinta.gradient(perdida, politica.trainable_variables)
    optimizador.apply_gradients(zip(grads, politica.trainable_variables))
    return float(perdida)

pesos = politica.get_weights()
perdida = paso_reinforce(retorno=+1.0)
print(f"retorno +1 -> pérdida {perdida:.4f}; π(a|s) después:", politica(estado).numpy().round(4)[0])

politica.set_weights(pesos)
perdida = paso_reinforce(retorno=-1.0)
print(f"retorno -1 -> pérdida {perdida:.4f}; π(a|s) después:", politica(estado).numpy().round(4)[0])
