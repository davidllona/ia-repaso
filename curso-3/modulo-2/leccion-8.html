21 · Autoencoder convolucional: decodificar con Conv2DTranspose o con UpSampling2D + Conv2D

Para qué sirve Entrenar dos autoencoders idénticos salvo el decodificador —uno con Conv2DTranspose y otro con UpSampling2D + Conv2D— sobre los dígitos de 8×8, y comparar su error. Es la aplicación más directa de la transpuesta que ya viste en los autoencoders del Curso 2.

import numpy as np
import keras
from keras import layers
from sklearn.datasets import load_digits
from sklearn.model_selection import train_test_split

digitos = load_digits()
X = (digitos.images / 16.0).astype("float32")[..., None]
Xtr, Xte = train_test_split(X, test_size=0.25, random_state=0)

def codificador():
    return [layers.Conv2D(8, 3, strides=2, padding="same", activation="relu"),    # 8x8 -> 4x4
            layers.Conv2D(4, 3, strides=2, padding="same", activation="relu")]    # 4x4 -> 2x2 (16 números)

def decodificador_transpuesto():
    return [layers.Conv2DTranspose(8, 3, strides=2, padding="same", activation="relu"),     # 2x2 -> 4x4
            layers.Conv2DTranspose(1, 3, strides=2, padding="same", activation="sigmoid")]  # 4x4 -> 8x8

def decodificador_resize():
    return [layers.UpSampling2D(2), layers.Conv2D(8, 3, padding="same", activation="relu"),   # 2x2 -> 4x4
            layers.UpSampling2D(2), layers.Conv2D(1, 3, padding="same", activation="sigmoid")]  # 4x4 -> 8x8

for nombre, decodificador in (("Conv2DTranspose", decodificador_transpuesto),
                              ("UpSampling2D + Conv2D", decodificador_resize)):
    keras.utils.set_random_seed(0)
    auto = keras.Sequential([keras.Input(shape=(8, 8, 1)), *codificador(), *decodificador()])
    auto.compile("adam", "mse")
    auto.fit(Xtr, Xtr, epochs=60, batch_size=32, verbose=0)
    mse = auto.evaluate(Xte, Xte, verbose=0)
    print(f"{nombre:22s} parámetros={auto.count_params():4d}  salida={auto.output_shape[1:]}  MSE en test={mse:.4f}")

print("MSE de predecir siempre la imagen media:", round(float(((Xte - Xtr.mean(axis=0)) ** 2).mean()), 4))

Ambos decodificadores tienen los mismos 741 parámetros y devuelven imágenes 8×8×1. El error de reconstrucción es parecido: 0,0277 con Conv2DTranspose y 0,0259 con UpSampling2D + Conv2D, los dos muy por debajo de 0,0734 de la imagen media (y eso con solo 16 números de cuello de botella). Con una sola semilla y una diferencia tan pequeña no se puede decir que uno sea mejor; en imágenes grandes y en generación, escalar + convolucionar evita el tablero de ajedrez (lo recomiendan Odena et al., 2016), y por eso es una opción frecuente.
