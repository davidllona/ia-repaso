1Fabricar la serie y cortarla en ventanas

Para qué sirve este paso: convertir una fila de 2.000 audiencias en 1.943 ejemplos «48 días → el día que viene 10 después», con un bucle hecho a mano para ver qué es una ventana y qué es el horizonte. Es la cuenta de la lección 1.

Paso 1 · Una serie sintética y sus ventanas, hechas a mano (fragmento 1)

Para qué sirve Fabricar la serie del módulo (tendencia + dos ondas + ruido) y convertirla en ejemplos (X, y) con un bucle de Python, para ver por qué salen 1.943.

import numpy as np

def crear_serie(n=2000, semilla=0):
    rng = np.random.default_rng(semilla)
    t = np.arange(n)
    tendencia = 0.0008 * t
    estacion = 1.0 * np.sin(2 * np.pi * t / 50) + 0.5 * np.sin(2 * np.pi * t / 12)
    ruido = rng.normal(0, 0.15, n)
    return (tendencia + estacion + ruido).astype("float32")

serie = crear_serie()
print("serie:", serie.shape, serie[:4].round(3))

VENTANA, HORIZONTE = 48, 10

def hacer_ventanas(serie, ventana, horizonte):
    X, y = [], []
    for i in range(len(serie) - ventana - horizonte + 1):
        X.append(serie[i : i + ventana])
        y.append(serie[i + ventana + horizonte - 1])
    return np.array(X)[..., None], np.array(y)

X, y = hacer_ventanas(serie, VENTANA, HORIZONTE)
print("X:", X.shape, " y:", y.shape)
print("ventanas esperadas:", len(serie) - VENTANA - HORIZONTE + 1)
print("X[0] acaba en el instante", VENTANA - 1, "-> y[0] es el instante", VENTANA + HORIZONTE - 1)
print("comprobación:", y[0] == serie[VENTANA + HORIZONTE - 1])

import keras
from keras import layers, ops

@keras.saving.register_keras_serializable(package="c3m3")
class PositionalEncoding(layers.Layer):
    def __init__(self, max_pasos, d_model, **kwargs):
        super().__init__(**kwargs)
        self.max_pasos, self.d_model = max_pasos, d_model
        pos = np.arange(max_pasos)[:, None]
        i = np.arange(d_model // 2)[None, :]
        angulos = pos / 10000 ** (2 * i / d_model)
        pe = np.zeros((max_pasos, d_model), dtype="float32")
        pe[:, 0::2], pe[:, 1::2] = np.sin(angulos), np.cos(angulos)
        self.pe = pe

    def call(self, x):
        pe = ops.convert_to_tensor(self.pe)
        return x + pe[: ops.shape(x)[1]]

    def get_config(self):
        return {**super().get_config(), "max_pasos": self.max_pasos, "d_model": self.d_model}

@keras.saving.register_keras_serializable(package="c3m3")
class TransformerBlock(layers.Layer):
    def __init__(self, d_model, num_heads, ff_dim, dropout=0.1, causal=False, **kwargs):
        super().__init__(**kwargs)
        self.d_model, self.num_heads, self.ff_dim = d_model, num_heads, ff_dim
        self.dropout, self.causal = dropout, causal
        self.att = layers.MultiHeadAttention(num_heads=num_heads, key_dim=d_model // num_heads)
        self.ffn = keras.Sequential([layers.Dense(ff_dim, activation="relu"), layers.Dense(d_model)])
        self.norm1 = layers.LayerNormalization(epsilon=1e-6)
        self.norm2 = layers.LayerNormalization(epsilon=1e-6)
        self.drop = layers.Dropout(dropout)

    def call(self, x, training=False):
        a = self.att(x, x, use_causal_mask=self.causal)
        x = self.norm1(x + self.drop(a, training=training))
        f = self.ffn(x)
        return self.norm2(x + self.drop(f, training=training))

    def get_config(self):
        return {**super().get_config(), "d_model": self.d_model, "num_heads": self.num_heads,
                "ff_dim": self.ff_dim, "dropout": self.dropout, "causal": self.causal}

def siguiente_token(logits, temperatura=1.0, top_k=None, rng=None):
    z = logits / temperatura
    if top_k is not None:
        corte = np.sort(z)[-top_k]
        z = np.where(z >= corte, z, -np.inf)
    p = softmax(z)
    return int(rng.choice(len(p), p=p)), p

import numpy as np

def softmax(z):
    z = z - z.max(axis=-1, keepdims=True)
    e = np.exp(z)
    return e / e.sum(axis=-1, keepdims=True)

def atencion(Q, K, V, mascara=None):
    dk = Q.shape[-1]
    puntos = Q @ K.T / np.sqrt(dk)
    if mascara is not None:
        puntos = np.where(mascara, puntos, -np.inf)
    pesos = softmax(puntos)
    return pesos @ V, pesos

import keras
from keras import layers, ops

@keras.saving.register_keras_serializable(package="c3m3")
class PositionalEncoding(layers.Layer):
    def __init__(self, max_pasos, d_model, **kwargs):
        super().__init__(**kwargs)
        self.max_pasos, self.d_model = max_pasos, d_model
        pos = np.arange(max_pasos)[:, None]
        i = np.arange(d_model // 2)[None, :]
        angulos = pos / 10000 ** (2 * i / d_model)
        pe = np.zeros((max_pasos, d_model), dtype="float32")
        pe[:, 0::2], pe[:, 1::2] = np.sin(angulos), np.cos(angulos)
        self.pe = pe

    def call(self, x):
        pe = ops.convert_to_tensor(self.pe)
        return x + pe[: ops.shape(x)[1]]

    def get_config(self):
        return {**super().get_config(), "max_pasos": self.max_pasos, "d_model": self.d_model}

@keras.saving.register_keras_serializable(package="c3m3")
class TransformerBlock(layers.Layer):
    def __init__(self, d_model, num_heads, ff_dim, dropout=0.1, causal=False, **kwargs):
        super().__init__(**kwargs)
        self.d_model, self.num_heads, self.ff_dim = d_model, num_heads, ff_dim
        self.dropout, self.causal = dropout, causal
        self.att = layers.MultiHeadAttention(num_heads=num_heads, key_dim=d_model // num_heads)
        self.ffn = keras.Sequential([layers.Dense(ff_dim, activation="relu"), layers.Dense(d_model)])
        self.norm1 = layers.LayerNormalization(epsilon=1e-6)
        self.norm2 = layers.LayerNormalization(epsilon=1e-6)
        self.drop = layers.Dropout(dropout)

    def call(self, x, training=False):
        a = self.att(x, x, use_causal_mask=self.causal)
        x = self.norm1(x + self.drop(a, training=training))
        f = self.ffn(x)
        return self.norm2(x + self.drop(f, training=training))

    def get_config(self):
        return {**super().get_config(), "d_model": self.d_model, "num_heads": self.num_heads,
                "ff_dim": self.ff_dim, "dropout": self.dropout, "causal": self.causal}

def siguiente_token(logits, temperatura=1.0, top_k=None, rng=None):
    z = logits / temperatura

        corte = np.sort(z)[-top_k]
        z = np.where(z >= corte, z, -np.inf)
    p = softmax(z)
    return int(rng.choice(len(p), p=p)), p

    rng = np.random.default_rng(semilla)
    t = np.linspace(0, 1, largo)
    etiquetas = rng.integers(0, 3, n)
    X = np.zeros((n, largo), dtype="float32")
    for i, c in enumerate(etiquetas):
        if c == 0:
            X[i] = np.sin(2 * np.pi * 2 * t + rng.uniform(0, 6.28))

        X[i] += rng.normal(0, 0.5, largo)

Ventanas a mano
serie[i : i + ventana] → serie[i + ventana + horizonte - 1].
timeseries_dataset_from_array
Ventanas por lotes; targets empieza en w + h − 1.
Normalizar sin fugas
Media y desviación solo de serie[:n_train].
Softmax estable
Restar el máximo, np.exp, dividir entre la suma.
Atención
softmax(Q @ K.T / np.sqrt(dk)) @ V.
Máscara causal
np.tril + np.where(m, puntos, -np.inf); en Keras, use_causal_mask=True.
MultiHeadAttention
num_heads, key_dim = d_model // num_heads; 4d² + 4d parámetros.
Posición
Senos en columnas pares (0::2), cosenos en impares; se suma.
Capa propia
class … (layers.Layer) con __init__, call, get_config y el decorador de registro.
Bloque
norm1(x + att) y norm2(x + ffn): 8.544 parámetros con d = 32.
Regresión
Dense(1) sin activación, loss="mse"; comparar con Ridge.
Mini-GPT
Ventanas de ids[:-1] y ids[1:], bloques causales, Dense(31) y from_logits=True.
Temperatura y top-k
logits / T; tapar con −∞ todo lo que quede por debajo de la k-ésima.

En tres líneas

  • Una serie se corta en el tiempo, se normaliza con train y se hace ventanas (ejemplos, pasos, rasgos); siempre hay que batir una línea base.
  • La atención es softmax(Q·Kᵀ/√dk)·V, con máscara causal para no mirar al futuro; con posición, atajos y LayerNorm forma el bloque Transformer.
  • El mismo bloque predice la audiencia (aunque una regresión lineal gana) y, con máscara causal, un mini-GPT que escribe con temperatura y top-k.
