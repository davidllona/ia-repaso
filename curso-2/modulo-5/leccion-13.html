Imagina que Netflix recibe miles de pósters y quiere saber dos cosas de cada uno: de qué género es (¿comedia o terror?) y cómo describirlo con una frase. El proyecto del curso es exactamente eso, pero con fotos de aviones: una aerolínea quiere saber qué tipo de daño hay en el fuselaje y cómo describirlo con palabras. ¿Cómo se resuelve con lo que ya sabes?

13 · Un dataset sintético de aviones con la estructura del proyecto

Para qué sirve Generar (sin internet) imágenes de «chapa de fuselaje» con una abolladura (dent) o una grieta (crack) y guardarlas en carpetas train/valid/test con una subcarpeta por clase, igual que las del laboratorio. Los fragmentos 14-18 usan estas carpetas.

import os
import numpy as np
from PIL import Image, ImageDraw

def chapa(rng, n=224):
    """Un trozo de fuselaje: gris azulado con degradado suave, grano y remaches."""
    y, x = np.mgrid[0:n, 0:n] / n
    brillo = 0.55 + 0.06 * rng.normal() + 0.1 * (x * rng.uniform(-1, 1) + y * rng.uniform(-1, 1))
    img = np.stack([brillo * 0.92, brillo * 0.97, brillo * 1.05], axis=-1)
    img += rng.normal(0, 0.025, (n, n, 1))
    for _ in range(rng.integers(3, 7)):                       # remaches
        cx, cy = rng.integers(10, n - 10, 2)
        img[cy - 2:cy + 2, cx - 2:cx + 2] -= 0.12
    return np.clip(img, 0, 1)

def con_abolladura(img, rng):
    """Dent: una zona clara junto a una oscura, como una chapa hundida iluminada de lado."""
    n = img.shape[0]
    cx, cy = rng.uniform(40, n - 40, 2)
    a, b = rng.uniform(14, 34, 2)
    th = rng.uniform(0, np.pi)
    yy, xx = np.mgrid[0:n, 0:n]
    u = (xx - cx) * np.cos(th) + (yy - cy) * np.sin(th)
    v = -(xx - cx) * np.sin(th) + (yy - cy) * np.cos(th)
    relieve = np.exp(-((u / a) ** 2 + (v / b) ** 2)) * (u / a) * rng.choice([-1, 1]) * rng.uniform(0.35, 0.6)
    return np.clip(img + relieve[..., None], 0, 1)

def con_grieta(img, rng):
    """Crack: una línea oscura que serpentea, con alguna ramificación."""
    n = img.shape[0]
    pil = Image.fromarray((img * 255).astype(np.uint8))
    dibujo = ImageDraw.Draw(pil)
    x, y = rng.uniform(30, n - 30, 2)
    ang = rng.uniform(0, 2 * np.pi)
    puntos = [(x, y)]
    for _ in range(rng.integers(10, 22)):
        ang += rng.normal(0, 0.45)
        x, y = x + 7 * np.cos(ang), y + 7 * np.sin(ang)
        puntos.append((x, y))
    dibujo.line(puntos, fill=(25, 25, 30), width=int(rng.integers(2, 5)))
    bx, by = puntos[rng.integers(2, len(puntos) - 2)]
    rama = [(bx, by), (bx + 18 * np.cos(ang + 0.9), by + 18 * np.sin(ang + 0.9))]
    dibujo.line(rama, fill=(30, 30, 35), width=2)
    return np.asarray(pil) / 255.0

def crear_dataset(raiz, tamanos, semilla=0):
    rng = np.random.default_rng(semilla)
    for conjunto, por_clase in tamanos.items():
        for clase, dañar in [("crack", con_grieta), ("dent", con_abolladura)]:
            carpeta = os.path.join(raiz, conjunto, clase)
            os.makedirs(carpeta, exist_ok=True)
            for i in range(por_clase):
                img = dañar(chapa(rng), rng)
                Image.fromarray((img * 255).astype(np.uint8)).save(f"{carpeta}/{clase}_{i:03d}.png")

crear_dataset("aviones_sinteticos", {"train": 150, "valid": 30, "test": 30})

for conjunto in ["train", "valid", "test"]:
    for clase in ["crack", "dent"]:
        carpeta = f"aviones_sinteticos/{conjunto}/{clase}"
        print(f"{carpeta:32s} {len(os.listdir(carpeta))} imágenes")
print("tamaño de una imagen:", Image.open("aviones_sinteticos/train/dent/dent_000.png").size)
