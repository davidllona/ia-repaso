17 · Los errores más típicos, con su mensaje real

Para qué sirve Provocar a propósito siete errores que verás de verdad al construir redes (formas, tipos, pérdidas mal usadas, capas no registradas) y leer el mensaje de cada uno.

import torch
from torch import nn

def probar(titulo, f):
    try:
        f()
        print(f"{titulo}: sin error")
    except Exception as e:
        print(f"{titulo}\n   {type(e).__name__}: {str(e).splitlines()[0]}")

torch.manual_seed(0)
x = torch.randn(4, 2)

# 1. La capa espera 3 características y le damos 2
probar("1. forma de la entrada", lambda: nn.Linear(3, 5)(x))

# 2. Entradas enteras (Long) en una capa de floats
probar("2. dtype de la entrada", lambda: nn.Linear(2, 5)(torch.tensor([[1, 2], [3, 4]])))

# 3. BCELoss necesita probabilidades, no logits
logits = torch.tensor([[2.0], [-1.0]])
etiquetas = torch.tensor([[1.0], [0.0]])
probar("3. BCELoss con logits", lambda: nn.BCELoss()(logits, etiquetas))

# 4. CrossEntropyLoss con etiquetas decimales de un vector
salida = torch.randn(4, 3)
probar("4. CrossEntropyLoss con etiquetas float", lambda: nn.CrossEntropyLoss()(salida, torch.tensor([0., 1., 2., 1.])))

# 5. Salida (N,1) y etiquetas (N,)
probar("5. BCE con formas (4,1) y (4,)", lambda: nn.BCELoss()(torch.rand(4, 1), torch.tensor([0., 1., 1., 0.])))

# 6. Capas guardadas en una lista de Python normal
class MalaRed(nn.Module):
    def __init__(self):
        super().__init__()
        self.capas = [nn.Linear(2, 3), nn.Linear(3, 1)]
    def forward(self, x):
        return self.capas[1](self.capas[0](x))
mala = MalaRed()
print("6. parámetros de una red con capas en una lista:", len(list(mala.parameters())))
probar("   ...y al crear el optimizador", lambda: torch.optim.SGD(mala.parameters(), lr=0.1))

# 7. Olvidar super().__init__()
class SinSuper(nn.Module):
    def __init__(self):
        self.linear = nn.Linear(2, 1)
probar("7. sin super().__init__()", SinSuper)

18 · Código de producción del curso original: MNIST con torchvision (no ejecutado aquí)

Para qué sirve Cómo se escribe la misma red con MNIST (60 000 imágenes de 28×28) en un ordenador con internet. Es el código que usa el curso oficial; aquí no se ha podido ejecutar porque necesita descargar los datos.

import torch
from torch import nn
from torch.utils.data import DataLoader
from torchvision import datasets, transforms

preparar = transforms.Compose([
    transforms.ToTensor(),
    transforms.Normalize((0.1307,), (0.3081,)),
])
datos_ent = datasets.MNIST(root="./datos", train=True, download=True, transform=preparar)
datos_val = datasets.MNIST(root="./datos", train=False, download=True, transform=preparar)
cargador_ent = DataLoader(datos_ent, batch_size=100, shuffle=True)
cargador_val = DataLoader(datos_val, batch_size=5000)

modelo = nn.Sequential(
    nn.Flatten(),
    nn.Linear(28 * 28, 100),
    nn.ReLU(),
    nn.Linear(100, 10),
)
criterio = nn.CrossEntropyLoss()
opt = torch.optim.SGD(modelo.parameters(), lr=0.1)

for epoca in range(10):
    for imagenes, etiquetas in cargador_ent:
        loss = criterio(modelo(imagenes), etiquetas)
        opt.zero_grad()
        loss.backward()
        opt.step()
    aciertos = 0
    with torch.no_grad():
        for imagenes, etiquetas in cargador_val:
            aciertos += (modelo(imagenes).argmax(dim=1) == etiquetas).sum().item()
    print(f"época {epoca + 1}: accuracy de validación {aciertos / len(datos_val):.4f}")
