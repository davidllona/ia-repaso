El fragmento 15 monta una red profunda configurable (6 capas ocultas de 64 neuronas, nn.ModuleList de bloques Linear → BatchNorm → ReLU → Dropout) y la entrena sobre los dígitos con seis recetas que añaden una pieza cada vez (media de 3 semillas, 30 épocas). Es una ablación: quitar o añadir una pieza y medir.

La lectura honesta: el gran salto (de 16 % a 97 %) lo da la combinación ReLU + inicialización He. Las recetas 1 y 2 no han aprendido nada en 30 épocas: sus gradientes son diminutos y la pérdida sigue en 2,30 (\( \ln10 \), elegir al azar entre 10 clases). A partir de ahí, BatchNorm, dropout y Adam son refinamientos: BatchNorm reduce el gradiente máximo de 10,2 a 3,9 (entrenamiento más estable) y el dropout baja un poco la pérdida de test, pero las diferencias de acierto (97,0, 97,1, 97,3, 96,7 %) están dentro del ruido de 3 semillas. En este problema pequeño, Adam no mejora a SGD con \( lr=0{,}1 \). Cuando la red es mucho más profunda, los datos más difíciles o los lotes más grandes, estas piezas pesan más. No hay receta universal: mide.

Receta de partida para una red densa profunda en PyTorch
  1. Activación ReLU (o una variante) y inicialización He con nn.init.kaiming_normal_; sesgos a cero.
  2. BatchNorm1d entre cada Linear y su activación si la red es profunda (y drop_last=True en el DataLoader).
  3. Optimizador: Adam (\( lr=10^{-3} \)) o SGD con momentum 0,9, más un planificador si el entrenamiento se estanca.
  4. Mide el gradiente por capa y la norma total (clip_grad_norm_ la devuelve); añade clipping si ves picos.
  5. Si hay sobreajuste (train ≫ test): dropout 0,1–0,5, AdamW con weight decay, más datos, parada temprana.
  6. Entrena en model.train(), evalúa en model.eval() con torch.no_grad().

1 · Red profunda con nn.ModuleList (y por qué no vale una lista normal)

Para qué sirve Construir una red de cualquier número de capas a partir de una lista de tamaños, contar sus parámetros y ver qué pasa si guardas las capas en una lista de Python en lugar de en nn.ModuleList.

import torch
import torch.nn as nn


class RedProfunda(nn.Module):
    def __init__(self, tamanos):
        super().__init__()
        self.capas = nn.ModuleList(
            [nn.Linear(tamanos[i], tamanos[i + 1]) for i in range(len(tamanos) - 1)]
        )

    def forward(self, x):
        for capa in self.capas[:-1]:
            x = torch.relu(capa(x))
        return self.capas[-1](x)


torch.manual_seed(0)
modelo = RedProfunda([64, 32, 32, 32, 10])
print(modelo)

total = 0
for nombre, p in modelo.named_parameters():
    print(f"{nombre:16s} {str(tuple(p.shape)):10s} {p.numel():5d}")
    total += p.numel()
print("parámetros totales:", total)
print("salida para un lote de 5:", tuple(modelo(torch.randn(5, 64)).shape))


def construir_mlp(tamanos):
    capas = []
    for i in range(len(tamanos) - 1):
        capas.append(nn.Linear(tamanos[i], tamanos[i + 1]))
        if i < len(tamanos) - 2:
            capas.append(nn.ReLU())
    return nn.Sequential(*capas)


otro = construir_mlp([64, 32, 32, 32, 10])
print("con Sequential:", sum(p.numel() for p in otro.parameters()), "parámetros")


class RedMala(nn.Module):
    def __init__(self):
        super().__init__()
        self.capas = [nn.Linear(4, 4), nn.Linear(4, 1)]   # lista normal de Python

    def forward(self, x):
        return self.capas[1](torch.relu(self.capas[0](x)))


mala = RedMala()
print("parámetros que ve RedMala:", len(list(mala.parameters())))
try:
    torch.optim.SGD(mala.parameters(), lr=0.1)
except ValueError as e:
    print("ValueError:", e)

15 · Receta completa: ablación con clip_grad_norm_

Para qué sirve Construir una red profunda configurable y añadir una a una las técnicas del módulo (ReLU, He, BatchNorm, dropout, Adam) para medir qué aporta cada una; mide además el mayor gradiente visto para introducir clip_grad_norm_.

import numpy as np
import torch
import torch.nn as nn
from sklearn.datasets import load_digits
from sklearn.model_selection import train_test_split
from torch.utils.data import DataLoader, TensorDataset

torch.set_num_threads(1)
X, y = load_digits(return_X_y=True)
Xtr, Xte, ytr, yte = train_test_split(X / 16.0, y, test_size=0.25, random_state=0, stratify=y)
Xtr, Xte = torch.tensor(Xtr, dtype=torch.float32), torch.tensor(Xte, dtype=torch.float32)
ytr, yte = torch.tensor(ytr), torch.tensor(yte)
perdida = nn.CrossEntropyLoss()


class RedProfunda(nn.Module):
    def __init__(self, bloques=6, ancho=64, act=nn.ReLU, bn=False, p=0.0, he=False):
        super().__init__()
        self.bloques = nn.ModuleList()
        for i in range(bloques):
            capas = [nn.Linear(64 if i == 0 else ancho, ancho, bias=not bn)]
            if bn:
                capas.append(nn.BatchNorm1d(ancho))
            capas.append(act())
            if p > 0:
                capas.append(nn.Dropout(p))
            self.bloques.append(nn.Sequential(*capas))
        self.salida = nn.Linear(ancho, 10)
        if he:
            for m in self.modules():
                if isinstance(m, nn.Linear):
                    nn.init.kaiming_normal_(m.weight, nonlinearity="relu")
                    if m.bias is not None:
                        nn.init.zeros_(m.bias)

    def forward(self, x):
        for bloque in self.bloques:
            x = bloque(x)
        return self.salida(x)


def entrenar(config, crear_opt, semilla, epocas=30):
    torch.manual_seed(semilla)
    modelo = RedProfunda(**config)
    opt = crear_opt(modelo.parameters())
    datos = DataLoader(TensorDataset(Xtr, ytr), batch_size=64, shuffle=True, drop_last=True,
                       generator=torch.Generator().manual_seed(semilla))
    norma_max = 0.0
    for _ in range(epocas):
        modelo.train()                                    # activa dropout y BatchNorm de entrenamiento
        for xb, yb in datos:
            opt.zero_grad()
            perdida(modelo(xb), yb).backward()
            norma = nn.utils.clip_grad_norm_(modelo.parameters(), max_norm=5.0)
            norma_max = max(norma_max, norma.item())
            opt.step()
    modelo.eval()                                         # dropout apagado, BatchNorm con medias móviles
    with torch.no_grad():
        return [perdida(modelo(Xtr), ytr).item(), (modelo(Xtr).argmax(1) == ytr).float().mean().item(),
                perdida(modelo(Xte), yte).item(), (modelo(Xte).argmax(1) == yte).float().mean().item(), norma_max]


sgd = lambda p: torch.optim.SGD(p, lr=0.1)
adam = lambda p: torch.optim.Adam(p, lr=1e-3)
recetas = [
    ("1. Sigmoide, init defecto, SGD", dict(act=nn.Sigmoid), sgd),
    ("2. ReLU, init defecto, SGD", dict(), sgd),
    ("3. ReLU + He, SGD", dict(he=True), sgd),
    ("4. + BatchNorm", dict(he=True, bn=True), sgd),
    ("5. + Dropout 0.2", dict(he=True, bn=True, p=0.2), sgd),
    ("6. + Adam (lr 1e-3)", dict(he=True, bn=True, p=0.2), adam),
]
print(f"{'receta':32s} {'pérd.train':>10s} {'acc.train':>9s} {'pérd.test':>9s} {'acc.test':>8s} {'‖g‖ máx':>8s}")
for nombre, config, crear_opt in recetas:
    r = np.mean([entrenar(config, crear_opt, s) for s in range(3)], axis=0)
    print(f"{nombre:32s} {r[0]:10.3f} {r[1]:9.3f} {r[2]:9.3f} {r[3]:8.3f} {r[4]:8.2f}")
