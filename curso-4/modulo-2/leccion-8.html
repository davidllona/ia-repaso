Errores típicos de Dataset y DataLoader

Cuando algo falla en la carga de datos el mensaje de PyTorch suele salir de las profundidades del DataLoader. ¿Cómo se reconoce cada error por su mensaje y cómo se arregla?

2 · TensorDataset: el Dataset ya hecho para tensores

Para qué sirve Cuando los datos ya son tensores en memoria, no hace falta escribir una clase: TensorDataset junta varios tensores del mismo tamaño y devuelve en cada ds[i] la fila i de cada uno.

import torch
from torch.utils.data import TensorDataset

X = torch.arange(12, dtype=torch.float32).reshape(6, 2)
y = torch.tensor([0, 1, 0, 1, 1, 0])

ds = TensorDataset(X, y)
print("len(ds)     :", len(ds))
print("ds[2]       :", ds[2])
print("ds[1:3]     :", ds[1:3])
print("ds.tensors  :", [tuple(t.shape) for t in ds.tensors])

try:
    TensorDataset(X, y[:5])
except AssertionError as error:
    print("tamaños distintos ->", type(error).__name__, ":", error)

7 · Estadísticas del dataset, Normalize y la versión moderna (v2)

Para qué sirve Calcular la media y la desviación de todos los píxeles de un dataset real (los dígitos de scikit-learn), usarlas en Normalize, comprobar que la imagen queda con media 0 y desviación 1, y ver la API moderna torchvision.transforms.v2.

import numpy as np
import torch
from PIL import Image
from sklearn.datasets import load_digits
from torchvision import transforms
from torchvision.transforms import v2

digitos = load_digits()
imagenes = np.round(digitos.images / 16 * 255).astype(np.uint8)
print("imágenes:", imagenes.shape, imagenes.dtype, " valores", imagenes.min(), "a", imagenes.max())

todas = torch.tensor(imagenes, dtype=torch.float32) / 255
media, desv = todas.mean().item(), todas.std().item()
print(f"media = {media:.4f}   desviación = {desv:.4f}   (de todos los píxeles)")

tuberia = transforms.Compose([transforms.ToTensor(), transforms.Normalize((media,), (desv,))])
t = tuberia(Image.fromarray(imagenes[0]))
print("una imagen:", tuple(t.shape), " min", round(t.min().item(), 3), " max", round(t.max().item(), 3))
lote = torch.stack([tuberia(Image.fromarray(im)) for im in imagenes])
print(f"todas tras Normalize: media {lote.mean().item():.4f}  desviación {lote.std().item():.4f}")

moderna = v2.Compose([v2.ToImage(), v2.ToDtype(torch.float32, scale=True), v2.Normalize((media,), (desv,))])
print("v2 da lo mismo que la clásica:", torch.allclose(moderna(Image.fromarray(imagenes[0])), t, atol=1e-6))

grande = transforms.Compose([transforms.Resize((16, 16)), transforms.CenterCrop(12), transforms.ToTensor()])
print("Resize 16x16 + CenterCrop 12 ->", tuple(grande(Image.fromarray(imagenes[0])).shape))

Con todo el conjunto transformado, la media es 0,0000 y la desviación 1,0000. La tubería v2 da lo mismo (True), y Resize((16,16)) + CenterCrop(12) devuelve un tensor (1, 12, 12).

9 · Dataset de imágenes con un CSV de etiquetas

Para qué sirve Leer imágenes desde archivos PNG cuyo nombre y clase están en un CSV (el formato del laboratorio de IBM): el Dataset abre el archivo solo cuando se le pide la muestra (carga perezosa) y le aplica la transformación.

Para tener algo que leer sin internet, el fragmento empieza generando 6 imágenes PNG de 10×10 en una carpeta temporal: las de clase 0 llevan un cuadrado brillante en el centro y las de clase 1 una barra horizontal, sobre ruido oscuro. Además escribe indice.csv con una fila por imagen.

import os, tempfile
import numpy as np
import pandas as pd
import torch
from PIL import Image
from torch.utils.data import Dataset, DataLoader
from torchvision import transforms

carpeta = tempfile.mkdtemp()
rng = np.random.default_rng(0)
filas = []
for k in range(6):
    clase = k % 2
    lienzo = rng.integers(0, 40, (10, 10)).astype(np.uint8)
    if clase == 0:
        lienzo[2:8, 2:8] = 220
    else:
        lienzo[4:6, :] = 220
    nombre = f"img_{k}.png"
    Image.fromarray(lienzo).save(os.path.join(carpeta, nombre))
    filas.append((nombre, clase))
pd.DataFrame(filas, columns=["archivo", "etiqueta"]).to_csv(os.path.join(carpeta, "indice.csv"), index=False)
print(open(os.path.join(carpeta, "indice.csv")).read().strip().splitlines()[:3])

class ImagenesCSV(Dataset):
    def __init__(self, carpeta, transform=None):
        self.carpeta = carpeta
        self.indice = pd.read_csv(os.path.join(carpeta, "indice.csv"))
        self.transform = transform

    def __len__(self):
        return len(self.indice)

    def __getitem__(self, i):
        fila = self.indice.iloc[i]
        imagen = Image.open(os.path.join(self.carpeta, fila["archivo"])).convert("L")
        if self.transform is not None:
            imagen = self.transform(imagen)
        return imagen, int(fila["etiqueta"])

ds = ImagenesCSV(carpeta, transform=transforms.ToTensor())
imagen, etiqueta = ds[0]
print("len:", len(ds), " ds[0]:", tuple(imagen.shape), imagen.dtype, " etiqueta:", etiqueta)
lote_x, lote_y = next(iter(DataLoader(ds, batch_size=4)))
print("lote:", tuple(lote_x.shape), lote_x.dtype, " etiquetas:", lote_y.tolist(), lote_y.dtype)

19 · Dataset de dígitos con transform, random_split y DataLoader

Para qué sirve Montar el equivalente local de «MNIST + transforms + validación»: un Dataset propio sobre los dígitos de scikit-learn que devuelve imágenes PIL transformadas, repartido en entrenamiento / validación / prueba con random_split y servido por un DataLoader.

import numpy as np
import torch
from PIL import Image
from sklearn.datasets import load_digits
from torch.utils.data import Dataset, DataLoader, random_split
from torchvision import transforms

class Digitos(Dataset):
    def __init__(self, transform=None):
        d = load_digits()
        self.imagenes = np.round(d.images / 16 * 255).astype(np.uint8)
        self.etiquetas = d.target
        self.transform = transform

    def __len__(self):
        return len(self.etiquetas)

    def __getitem__(self, i):
        imagen = Image.fromarray(self.imagenes[i])
        if self.transform is not None:
            imagen = self.transform(imagen)
        return imagen, int(self.etiquetas[i])

ds = Digitos(transform=transforms.Compose([transforms.ToTensor(), transforms.Normalize((0.3053,), (0.3759,))]))
print("total:", len(ds))
imagen, etiqueta = ds[0]
print("ds[0]:", tuple(imagen.shape), imagen.dtype, "etiqueta", etiqueta)

generador = torch.Generator().manual_seed(0)
entrenamiento, validacion, prueba = random_split(ds, [0.7, 0.15, 0.15], generator=generador)
print("reparto:", len(entrenamiento), len(validacion), len(prueba), " suma =", len(entrenamiento) + len(validacion) + len(prueba))
print("primeros índices de entrenamiento:", entrenamiento.indices[:5])

cargador = DataLoader(entrenamiento, batch_size=64, shuffle=True, generator=torch.Generator().manual_seed(0))
print("lotes por época:", len(cargador), " (el último tiene", len(entrenamiento) - 64 * (len(cargador) - 1), "imágenes)")
lote_x, lote_y = next(iter(cargador))
print("primer lote:", tuple(lote_x.shape), lote_x.dtype, "| etiquetas", tuple(lote_y.shape), lote_y.dtype)
print("etiquetas del lote:", lote_y[:12].tolist())

22 · Los cinco errores típicos de Dataset y DataLoader, con su mensaje real

Para qué sirve Provocar a propósito los errores más frecuentes y leer el mensaje que da PyTorch, para reconocerlos la primera vez que te aparezcan en un proyecto.

import torch
from PIL import Image
from torch.utils.data import Dataset, DataLoader

class SinLen:
    def __getitem__(self, i):
        return torch.tensor([float(i)])

class Tamanos(Dataset):
    def __len__(self):
        return 4
    def __getitem__(self, i):
        return torch.zeros(1, 4 + i, 4)

class ConPIL(Dataset):
    def __len__(self):
        return 4
    def __getitem__(self, i):
        return Image.new("L", (4, 4)), i

def intentar(nombre, funcion):
    try:
        funcion()
    except Exception as error:
        print(f"{nombre}\n   {type(error).__name__}: {str(error).splitlines()[0]}")

intentar("1) Dataset sin __len__", lambda: next(iter(DataLoader(SinLen(), batch_size=2, shuffle=True))))
intentar("2) imágenes de tamaños distintos", lambda: next(iter(DataLoader(Tamanos(), batch_size=2))))
intentar("3) imagen PIL sin ToTensor", lambda: next(iter(DataLoader(ConPIL(), batch_size=2))))
intentar("4) next() sobre el DataLoader", lambda: next(DataLoader(Tamanos(), batch_size=2)))
intentar("5) float32 por float64", lambda: torch.mm(torch.ones(2, 3), torch.ones(3, 2, dtype=torch.float64)))
