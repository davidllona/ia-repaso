20 · Guardar y cargar el modelo con state_dict

Para qué sirve Guarda solo los números del modelo, crea otro modelo de la misma arquitectura, le carga esos números y comprueba que predice exactamente lo mismo.

import os
import torch

torch.save(modelo.state_dict(), "digitos_softmax.pt")
print("claves guardadas:", list(modelo.state_dict().keys()))
print("tamaño del archivo:", os.path.getsize("digitos_softmax.pt"), "bytes")

nuevo = RegresionSoftmax(64, 10)
nuevo.load_state_dict(torch.load("digitos_softmax.pt"))
nuevo.eval()
with torch.no_grad():
    print("¿mismos logits que el modelo original?", torch.equal(nuevo(X_te), modelo(X_te)))
    print("accuracy del modelo cargado:", round((nuevo(X_te).argmax(dim=1) == y_te).float().mean().item(), 4))
os.remove("digitos_softmax.pt")

23 · Entrenar con MNIST real (no ejecutado aquí)

Para qué sirve El mismo modelo y el mismo bucle sobre las 60.000 imágenes de MNIST descargadas con torchvision. No se ha ejecutado: el entorno donde se redactó este módulo no tiene acceso a internet (la descarga devolvió un error 403).

import torch
from torch import nn
from torch.utils.data import DataLoader
from torchvision import datasets, transforms

torch.manual_seed(42)
transformacion = transforms.ToTensor()
entrenamiento = datasets.MNIST("datos", train=True, download=True, transform=transformacion)
prueba = datasets.MNIST("datos", train=False, download=True, transform=transformacion)
cargador_tr = DataLoader(entrenamiento, batch_size=64, shuffle=True)
cargador_te = DataLoader(prueba, batch_size=1000)

modelo = nn.Sequential(nn.Flatten(), nn.Linear(28 * 28, 10))
criterio = nn.CrossEntropyLoss()
optimizador = torch.optim.SGD(modelo.parameters(), lr=0.1)

for epoca in range(1, 6):
    modelo.train()
    for xb, yb in cargador_tr:
        optimizador.zero_grad()
        perdida = criterio(modelo(xb), yb)
        perdida.backward()
        optimizador.step()

    modelo.eval()
    aciertos = 0
    with torch.no_grad():
        for xb, yb in cargador_te:
            aciertos += (modelo(xb).argmax(dim=1) == yb).sum().item()
    print(f"época {epoca} | accuracy test {aciertos / len(prueba):.4f}")
